# Example API Usage
import requests

api_key = "YOUR_API_KEY"
headers = {"Authorization": f"Bearer {api_key}"}

# Get content
response = requests.get(
    "https://yourdomain.com/api/content",
    headers=headers
)

# Create content
data = {
    "account": "Affiliate 1",
    "category": "Ide",
    "text": "Your content here"
}
response = requests.post(
    "https://yourdomain.com/api/content",
    headers=headers,
    json=data
)
